function wireTaskRowEvents(el, code, viewRole, deptUsers, subType, isDelegatedView){ el.querySelectorAll('.task-row').forEach(row=>{ const id = row.dataset.id; const task = TASKS.find(t=>String(t.id) === String(id)); if(!task) return; const select = row.querySelector('[data-role="status"]'); if (select) { select.addEventListener('change', ()=>{ select.className = 'status-select status-' + select.value.replace(/\s+/g,'-'); }); } const saveBtn = row.querySelector('[data-role="save"]'); if (saveBtn) { saveBtn.addEventListener('click', async ()=>{ let newStatus = select ? select.value : 'Not Done'; let newNotes = ''; if (isTargetCat(task.category)) { const achieveInput = row.querySelector('[data-role="achieve-input"]'); newNotes = achieveInput ? achieveInput.value : ''; const targetScore = parseFloat(task.score) || 0; const achievedVal = parseFloat(newNotes) || 0; if (achievedVal >= targetScore && targetScore > 0) { newStatus = 'Done'; } else { newStatus = 'Not Done'; } } else { const noteInput = row.querySelector('[data-role="notes"]'); newNotes = noteInput ? noteInput.value : ''; } const vDateToSave = row.dataset.vdate || selectedLogDate; // Await the save result and strictly check for success before reloading const success = await saveSingleTaskLog(task, newStatus, newNotes, row, deptUsers, isDelegatedView, vDateToSave); if (success) { showToast('Saved Successfully!'); await loadData(); renderDeptTasks(document.getElementById('content'), code, viewRole, subType); } }); } if(viewRole !== 'employee') { const delBtn = row.querySelector('[data-role="delete"]'); if (delBtn) { delBtn.addEventListener('click', async ()=>{ if(!confirm('Remove this task from the master list?')) return; const { error } = await supabaseClient.from('tasks').delete().eq('id', id); if(error){ alert('DATABASE ERROR deleting task: ' + error.message); } else { TASKS = TASKS.filter(t=>String(t.id) !== String(id)); showToast('Task removed'); renderDeptTasks(el, code, viewRole, subType); } }); } } }); } async function saveSingleTaskLog(task, newStatus, newNotes, row, deptUsers, isDelegatedView, explicitVDate) { const assignSelect = row.querySelector('[data-role="assign"]'); let currentAssignee = task.assigned_to; if (assignSelect && assignSelect.value && assignSelect.value !== 'DISTRIBUTE_ALL') { currentAssignee = assignSelect.value; if (currentAssignee !== task.assigned_to) { const assignedUser = USERS.find(u => u && u.name === currentAssignee); await supabaseClient.from('tasks').update({ assigned_to: currentAssignee, mobile: assignedUser ? assignedUser.mobile : null, updated_by: currentUser.name, updated_at: new Date().toISOString() }).eq('id', task.id); task.assigned_to = currentAssignee; } } if (task.frequency === 'One-Time') { const { error } = await supabaseClient.from('tasks').update({ status: newStatus, notes: newNotes, updated_by: currentUser.name, updated_at: new Date().toISOString() }).eq('id', task.id); if (error) { alert("Database Error saving task: " + error.message); return false; } task.status = newStatus; task.notes = newNotes; return true; } else { const activeDate = explicitVDate || (isDelegatedView || activeTab === 'mypending' ? getTodayStr() : selectedLogDate); // Safely check if a log already exists to avoid strict onConflict constraint requirements const { data: existingLog } = await supabaseClient .from('task_logs') .select('id') .eq('task_id', task.id) .eq('log_date', activeDate) .maybeSingle(); let saveErr; if (existingLog) { const { error } = await supabaseClient.from('task_logs').update({ status: newStatus, notes: newNotes, assigned_to: currentAssignee, updated_by: currentUser.name, updated_at: new Date().toISOString() }).eq('id', existingLog.id); saveErr = error; } else { const { error } = await supabaseClient.from('task_logs').insert([{ task_id: task.id, assigned_to: currentAssignee, log_date: activeDate, status: newStatus, notes: newNotes, updated_by: currentUser.name, updated_at: new Date().toISOString() }]); saveErr = error; } if (saveErr) { alert("Database Error updating daily log: " + saveErr.message); return false; } // Only update master row cache if it was for today if (activeDate === getTodayStr()) { task.status = newStatus; task.notes = newNotes; await supabaseClient.from('tasks').update({ status: newStatus, notes: newNotes, updated_by: currentUser.name, updated_at: new Date().toISOString() }).eq('id', task.id); } return true; } } async function saveAllTasksForDay(el, code, viewRole, isDelegatedView) { const rows = el.querySelectorAll('.task-row'); if (rows.length === 0) return; const baseActiveDate = (isDelegatedView || activeTab === 'mypending') ? getTodayStr() : selectedLogDate; let successCount = 0; for (let i = 0; i < rows.length; i++) { const row = rows[i]; const id = row.dataset.id; const task = TASKS.find(t => String(t.id) === String(id)); if (!task) continue; const select = row.querySelector('[data-role="status"]'); let newStatus = select ? select.value : 'Not Done'; let newNotes = ''; if (isTargetCat(task.category)) { const achieveInput = row.querySelector('[data-role="achieve-input"]'); newNotes = achieveInput ? achieveInput.value : ''; const targetScore = parseFloat(task.score) || 0; const achievedVal = parseFloat(newNotes) || 0; if (achievedVal >= targetScore && targetScore > 0) newStatus = 'Done'; else newStatus = 'Not Done'; } else { const noteInput = row.querySelector('[data-role="notes"]'); newNotes = noteInput ? noteInput.value : ''; } if (task.frequency === 'One-Time') { const { error } = await supabaseClient.from('tasks').update({ status: newStatus, notes: newNotes, updated_by: currentUser.name, updated_at: new Date().toISOString() }).eq('id', task.id); if (!error) { task.status = newStatus; task.notes = newNotes; successCount++; } } else { const rowVDate = row.dataset.vdate || baseActiveDate; const { data: existingLog } = await supabaseClient.from('task_logs') .select('id').eq('task_id', task.id).eq('log_date', rowVDate).maybeSingle(); let saveErr; if (existingLog) { const { error } = await supabaseClient.from('task_logs').update({ status: newStatus, notes: newNotes, updated_by: currentUser.name, updated_at: new Date().toISOString() }).eq('id', existingLog.id); saveErr = error; } else { const { error } = await supabaseClient.from('task_logs').insert([{ task_id: task.id, assigned_to: task.assigned_to, log_date: rowVDate, status: newStatus, notes: newNotes, updated_by: currentUser.name, updated_at: new Date().toISOString() }]); saveErr = error; } if (!saveErr) { successCount++; if (rowVDate === getTodayStr()) { task.status = newStatus; task.notes = newNotes; supabaseClient.from('tasks').update({ status: newStatus, notes: newNotes, updated_by: currentUser.name, updated_at: new Date().toISOString() }).eq('id', task.id).then(); } } } } showToast(`Successfully saved ${successCount} tasks!`); await loadData(); renderContent(); }
